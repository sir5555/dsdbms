create or replace function maxmin(int , int) returns text as '
declare 
	result text := ''not found'';
	n1 alias for $1;
	n2 alias for $2;
begin 
	if n1 = n2 then 
		result := ''Both numbers are same'';
	elsif n1>n2 then 
		result := n1||'' is greater than ''||n2;
	else	
		result := n2||'' is greater than ''||n1;
	end if ;
	return result;
end;'
language 'plpgsql';

create or replace function sum_avg(n int) returns text as'
declare
	i int := 1;
	sum int :=0;
	avg numeric(6,2);
	result text := '''';
begin
	while i<=n loop
		sum :=sum+i;
		i := i+1;
	end loop;
	avg := sum*1.0/n;
	result := '' sum = ''|| sum|| '',Average = '' || avg;
	return result;
end;'
language 'plpgsql';

declare 
	i int;
	cnt int :=0;

create or replace function num_word(n int) returns text as '
declare
	result text := '''';
	d int;
begin
	while n>0 loop
		d := n%10;
		case d
		    when 0 then result := ''Zero ''||result;
		    when 1 then result := ''One ''||result;
		    when 2 then result := ''Two ''||result;
		    when 3 then result := ''Three ''||result;
		    when 4 then result := ''Four ''||result;
		    when 5 then result := ''Five ''||result;
		    when 6 then result := ''Six ''||result;
		    when 7 then result := ''Seven ''||result;
		    when 8 then result := ''Eight ''||result;
		    when 9 then result := ''Nine ''||result;
		end case;
		n:=n/10;
	end loop;
	return result;
end;'
language 'plpgsql';
