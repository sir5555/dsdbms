create or replace function routeofbus(bnum int) returns text as '
declare
	rec record;
	result text :='' '';
	c1 cursor for 
	select * from route1 r , brt br
	where r.rno = br.rno and br.bno= bnum;

begin 
	open c1;
	loop
	fetch c1 into rec;
	exit when not found;

	result := result || '' routeno:''|| rec.rno||'' from : ''||rec.source||'' To : ''||rec.dest||'' stations : ''||rec.nos||E''\n'';
	end loop;
	close c1;
	return result;
	end;'
	language 'plpgsql';

create or replace function busbtw(src varchar,des varchar) returns text as '
declare 
	rec record;
	result text := '' '';
	c2 cursor for 
	select * from bus1 b , route1 r , brt br
	where b.bno = br.bno and r.rno = br.rno and source = src and dest = des;

begin 
	open c2;
	loop
	fetch c2 into rec;
	exit when not found;
	result := result||''bus no : ''||rec.bno||'' capacity: ''||rec.cap||'' depot : ''||rec.depot_name||E''\n'';
	end loop;
	close c2;
	return result;
	end;'
	language 'plpgsql';

create or replace function disp(int)returns text as'
declare
		mon alias for $1;
		cur1 cursor for select pname,bdt
		from person 
		where extract(month from bdt)=mon;
		prec record;
		result text :=''Person name    Birth date'';
begin
		open cur1;
		loop
			fetch cur1 into prec;
			exit when NOT FOUND ;
		result:=result||E''\n''||prec.pname||''               ''||prec.bdt;
		end loop;
		close cur1;
		return result;
end;
'language 'plpgsql';

create or replace function up(varchar(15))returns text as'
declare
		aa alias for $1;
		cur2 cursor for select pname
		from person p,area a
		where p.pno=a.pno and atype=aa; 
		pnm person.pname%type;
		result text :=''Person name    Area type'';
begin
		open cur2;
		loop
			fetch cur2 into pnm;
			exit when NOT FOUND ;
		result:=result||E''\n''||pnm||''               ''||aa;
		end loop;
		close cur2;
		return result;
end;
'language 'plpgsql';
