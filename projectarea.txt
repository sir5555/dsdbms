create or replace function pdt(varchar(20))returns text as'
declare
    prec record;
    ecnt int:=0;
    result text:=''Project name  Project no  Employee name  start date  Total employee'';
begin
          select count(*) into ecnt
          from project
          where pname=$1;
          if NOT FOUND then
              raise exception''Project name % is Invalid!!'',$1;
          end if;
          ecnt := 0;
          for prec in
	        select p.pno,pname,ename,sdate
	        from project p,empl e,pe
	        where p.pno=pe.pno and e.eid=pe.eid and pname=$1 
          loop
	        result := result ||E''\n''||prec.pname||''            ''||prec.pno||''             ''||prec.ename||''        ''||prec.sdate;
	        ecnt := ecnt+1;
          end loop;
          result := result || E''\n Number of employees working on this project : "||ecnt;
          return result;

end;
'language'plpgsql';

create or replace function empdt(int)returns int as'
declare
    eno alias for $1;
    pcnt int:=0;
begin
     select count(*) into pcnt
     from empl e,pe
     where e.eid=pe.eid and e.eid=eno;
     if pcnt=0 then
          raise exception''Employee number % is Invalid!!'',eno;
     else
          raise notice'' The number of project of employee having eid % = %'',eno,pcnt;
     end if;
     return pcnt;
end;
'language'plpgsql';

create or replace function apdt(varchar(15))returns text as'
declare
    cnt int :=0;
    anm alias for $1;
    prec record;
    result text:=''Area name   Person name  Person no   Birth date      Income  '';
begin
          select count(*) into cnt
          from area
          where aname=$1;
          if cnt=0 then
              raise exception''Area name  % is Invalid!!'',$1;
          end if;
          for prec in
	        select p.pno,aname,pname,bdt,income
	        from person p,area a
	        where p.pno=a.pno and aname=$1 
          loop
	        result := result ||E''\n''||prec.aname||''          ''||prec.pname||''         ''||prec.pno||''       ''||prec.bdt||''        ''||prec.income;
          end loop;
          return result;
end;
'language'plpgsql';
