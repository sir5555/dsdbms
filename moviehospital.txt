CREATE OR REPLACE FUNCTION check_budget()
RETURNS TRIGGER AS '
BEGIN
    IF NEW.budget < 100000 THEN
        RAISE EXCEPTION ''Budget must be greater than 1,00,000'';
    END IF;
    RETURN NEW;
END;
'LANGUAGE 'plpgsql';
CREATE TRIGGER trg_check_budget
BEFORE INSERT ON Movie
FOR EACH ROW
EXECUTE PROCEDURE check_budget();

begin 
	if new.charges<0 then 
		raise exception ''charges should not be negative'';
	end if ;
	return new;
end ;'
